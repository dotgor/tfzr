RESENJE - POGLEDI
------------------------------------------
CREATE VIEW `Fakultet_VP_2025`.`SviPodacioStudentima` AS select `STUDENT`.`BrojIndeksa`, `STUDENT`.`Prezime`, `STUDENT`.`Ime`, `SMER`.`Naziv` as `NazivSmera` from `Fakultet_VP_2025`.`STUDENT` INNER JOIN `Fakultet_VP_2025`.`SMER` ON `Fakultet_VP_2025`.`STUDENT`.`OznakaSmera`=`Fakultet_VP_2025`.`SMER`.`Oznaka`;
CREATE VIEW `Fakultet_VP_2025`.`Listing` AS SELECT red.id, polazni.naziv as polazni_grad, odredisni.naziv as odredisni_grad, red.vreme, red.cena FROM `Fakultet_VP_2025`.`red_voznje` AS red INNER JOIN `gradovi` polazni ON polazni.id = red.iz INNER JOIN `gradovi` odredisni ON odredisni.id = red.ka;


RESENJE ZA STORED PROCEDURU - unos reda voznje
-----------------------
USE `Fakultet_VP_2025`;
DROP procedure IF EXISTS `DodajRedVoznje`;
DELIMITER $$
USE `Fakultet_VP_2025`$$
CREATE PROCEDURE `DodajRedVoznje` (IN IzParametar int unsigned, IN KaParametar int unsigned, IN VremeParametar time, IN CenaParametar decimal(10,2))
BEGIN
INSERT INTO `red_voznje` (`iz`, `ka`, `vreme`, `cena`) VALUES (IzParametar, KaParametar, VremeParametar, CenaParametar);
END
$$
DELIMITER ;
